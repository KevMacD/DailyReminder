from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime, timezone
from typing import Any, Dict, List, Optional


# -----------------------------
# Helpers
# -----------------------------
def _req(d: Dict[str, Any], key: str) -> Any:
    if key not in d:
        raise KeyError(f"Missing required key: {key!r}")
    return d[key]


def _dt_utc(ts: int) -> datetime:
    # OpenWeather "dt" values are UNIX seconds; interpret as UTC
    return datetime.fromtimestamp(int(ts), tz=timezone.utc)


def _clamp(name: str, value: float, lo: float, hi: float) -> float:
    if not (lo <= value <= hi):
        raise ValueError(f"{name} out of range: {value} (expected {lo}..{hi})")
    return value


def _int_range(name: str, value: int, lo: int, hi: int) -> int:
    if not (lo <= value <= hi):
        raise ValueError(f"{name} out of range: {value} (expected {lo}..{hi})")
    return value


def k_to_c(k: float) -> float:
    return float(k) - 273.15


def k_to_f(k: float) -> float:
    return (float(k) - 273.15) * 9.0 / 5.0 + 32.0

@dataclass(frozen=True, slots=True)
class WeatherCondition:
    id: int
    main: str
    description: str
    icon: str

    @classmethod
    def from_dict(cls, d: Dict[str, Any]) -> "WeatherCondition":
        return cls(
            id=int(_req(d, "id")),
            main=str(_req(d, "main")),
            description=str(_req(d, "description")),
            icon=str(_req(d, "icon")),
        )


@dataclass(frozen=True, slots=True)
class CurrentWeather:
    dt: datetime
    sunrise: datetime
    sunset: datetime
    temp_k: float
    feels_like_k: float
    pressure_hpa: int
    humidity_pct: int
    dew_point_k: float
    uvi: float
    clouds_pct: int
    visibility_m: int
    wind_speed_mps: float
    wind_deg: int
    weather: List[WeatherCondition]

    @classmethod
    def from_dict(cls, d: Dict[str, Any]) -> "CurrentWeather":
        humidity = int(_req(d, "humidity"))
        clouds = int(_req(d, "clouds"))
        vis = int(_req(d, "visibility"))
        wind_deg = int(_req(d, "wind_deg"))
        uvi = float(_req(d, "uvi"))

        return cls(
            dt=_dt_utc(_req(d, "dt")),
            sunrise=_dt_utc(_req(d, "sunrise")),
            sunset=_dt_utc(_req(d, "sunset")),
            temp_k=float(_req(d, "temp")),
            feels_like_k=float(_req(d, "feels_like")),
            pressure_hpa=int(_req(d, "pressure")),
            humidity_pct=_int_range("humidity", humidity, 0, 100),
            dew_point_k=float(_req(d, "dew_point")),
            uvi=_clamp("uvi", uvi, 0.0, 25.0),  # typical practical bounds
            clouds_pct=_int_range("clouds", clouds, 0, 100),
            visibility_m=_int_range("visibility", vis, 0, 100000),
            wind_speed_mps=float(_req(d, "wind_speed")),
            wind_deg=_int_range("wind_deg", wind_deg, 0, 360),
            weather=[WeatherCondition.from_dict(x) for x in _req(d, "weather")],
        )

    # Kelvin -> Celsius/Fahrenheit convenience properties
    @property
    def temp_c(self) -> float:
        return k_to_c(self.temp_k)

    @property
    def temp_f(self) -> float:
        return k_to_f(self.temp_k)

    @property
    def feels_like_c(self) -> float:
        return k_to_c(self.feels_like_k)

    @property
    def feels_like_f(self) -> float:
        return k_to_f(self.feels_like_k)

    @property
    def dew_point_c(self) -> float:
        return k_to_c(self.dew_point_k)

    @property
    def dew_point_f(self) -> float:
        return k_to_f(self.dew_point_k)


@dataclass(frozen=True, slots=True)
class MinutelyForecast:
    dt: datetime
    precipitation_mm: float  # OpenWeather returns mm for the minute

    @classmethod
    def from_dict(cls, d: Dict[str, Any]) -> "MinutelyForecast":
        p = float(_req(d, "precipitation"))
        if p < 0:
            raise ValueError(f"precipitation cannot be negative: {p}")
        return cls(dt=_dt_utc(_req(d, "dt")), precipitation_mm=p)


@dataclass(frozen=True, slots=True)
class HourlyForecast:
    dt: datetime
    temp_k: float
    feels_like_k: float
    pressure_hpa: int
    humidity_pct: int
    dew_point_k: float
    uvi: float
    clouds_pct: int
    visibility_m: int
    wind_speed_mps: float
    wind_deg: int
    wind_gust_mps: Optional[float]
    weather: List[WeatherCondition]
    pop: float  # probability of precipitation (0..1)

    @classmethod
    def from_dict(cls, d: Dict[str, Any]) -> "HourlyForecast":
        humidity = int(_req(d, "humidity"))
        clouds = int(_req(d, "clouds"))
        vis = int(_req(d, "visibility"))
        wind_deg = int(_req(d, "wind_deg"))
        uvi = float(_req(d, "uvi"))
        pop = float(_req(d, "pop"))

        wind_gust = d.get("wind_gust", None)
        if wind_gust is not None:
            wind_gust = float(wind_gust)
            if wind_gust < 0:
                raise ValueError(f"wind_gust cannot be negative: {wind_gust}")

        return cls(
            dt=_dt_utc(_req(d, "dt")),
            temp_k=float(_req(d, "temp")),
            feels_like_k=float(_req(d, "feels_like")),
            pressure_hpa=int(_req(d, "pressure")),
            humidity_pct=_int_range("humidity", humidity, 0, 100),
            dew_point_k=float(_req(d, "dew_point")),
            uvi=_clamp("uvi", uvi, 0.0, 25.0),
            clouds_pct=_int_range("clouds", clouds, 0, 100),
            visibility_m=_int_range("visibility", vis, 0, 100000),
            wind_speed_mps=float(_req(d, "wind_speed")),
            wind_deg=_int_range("wind_deg", wind_deg, 0, 360),
            wind_gust_mps=wind_gust,
            weather=[WeatherCondition.from_dict(x) for x in _req(d, "weather")],
            pop=_clamp("pop", pop, 0.0, 1.0),
        )

    @property
    def temp_c(self) -> float:
        return k_to_c(self.temp_k)

    @property
    def temp_f(self) -> float:
        return k_to_f(self.temp_k)

    @property
    def feels_like_c(self) -> float:
        return k_to_c(self.feels_like_k)

    @property
    def feels_like_f(self) -> float:
        return k_to_f(self.feels_like_k)

    @property
    def dew_point_c(self) -> float:
        return k_to_c(self.dew_point_k)

    @property
    def dew_point_f(self) -> float:
        return k_to_f(self.dew_point_k)

@dataclass(frozen=True, slots=True)
class DailyForecast:
    dt: datetime
    sunrise: datetime
    sunset: datetime
    moonrise: Optional[datetime]
    moonset: Optional[datetime]
    moon_phase: float
    summary: str
    temp: DailyTemp
    feels_like: DailyFeelsLike
    pressure_hpa: int
    humidity_pct: int
    dew_point_k: float
    wind_speed_mps: float
    wind_deg: int
    wind_gust_mps: Optional[float]
    weather: List[WeatherCondition]
    clouds_pct: int
    pop: float
    uvi: float
    rain_mm: Optional[float]

    @classmethod
    def from_dict(cls, d: Dict[str, Any]) -> "DailyForecast":
        humidity = int(_req(d, "humidity"))
        clouds = int(_req(d, "clouds"))
        wind_deg = int(_req(d, "wind_deg"))
        pop = float(_req(d, "pop"))
        uvi = float(_req(d, "uvi"))
        moonrise_raw = _req(d, "moonrise")
        moonset_raw = _req(d, "moonset")

        # Some APIs use 0 to indicate "no event"
        moonrise = None if int(moonrise_raw) == 0 else _dt_utc(moonrise_raw)
        moonset = None if int(moonset_raw) == 0 else _dt_utc(moonset_raw)

        wind_gust = d.get("wind_gust", None)
        if wind_gust is not None:
            wind_gust = float(wind_gust)
            if wind_gust < 0:
                raise ValueError(f"wind_gust cannot be negative: {wind_gust}")

        rain = d.get("rain", None)
        if rain is not None:
            rain = float(rain)
            if rain < 0:
                raise ValueError(f"rain cannot be negative: {rain}")

        return cls(
            dt=_dt_utc(_req(d, "dt")),
            sunrise=_dt_utc(_req(d, "sunrise")),
            sunset=_dt_utc(_req(d, "sunset")),
            moonrise=moonrise,
            moonset=moonset,
            moon_phase=_clamp("moon_phase", float(_req(d, "moon_phase")), 0.0, 1.0),
            summary=str(_req(d, "summary")),
            temp=DailyTemp.from_dict(_req(d, "temp")),
            feels_like=DailyFeelsLike.from_dict(_req(d, "feels_like")),
            pressure_hpa=int(_req(d, "pressure")),
            humidity_pct=_int_range("humidity", humidity, 0, 100),
            dew_point_k=float(_req(d, "dew_point")),
            wind_speed_mps=float(_req(d, "wind_speed")),
            wind_deg=_int_range("wind_deg", wind_deg, 0, 360),
            wind_gust_mps=wind_gust,
            weather=[WeatherCondition.from_dict(x) for x in _req(d, "weather")],
            clouds_pct=_int_range("clouds", clouds, 0, 100),
            pop=_clamp("pop", pop, 0.0, 1.0),
            uvi=_clamp("uvi", uvi, 0.0, 25.0),
            rain_mm=rain,
        )

    @property
    def dew_point_c(self) -> float:
        return k_to_c(self.dew_point_k)

    @property
    def dew_point_f(self) -> float:
        return k_to_f(self.dew_point_k)


@dataclass(frozen=True, slots=True)
class OneCallResponse:
    lat: float
    lon: float
    timezone: str
    timezone_offset_sec: int
    current: CurrentWeather
    minutely: List[MinutelyForecast]
    hourly: List[HourlyForecast]
    daily: List[DailyForecast]

    @classmethod
    def from_dict(cls, d: Dict[str, Any]) -> "OneCallResponse":
        tz_off = int(_req(d, "timezone_offset"))
        # timezone_offset is seconds from UTC; practical bounds are -14h..+14h
        _int_range("timezone_offset", tz_off, -14 * 3600, 14 * 3600)

        return cls(
            lat=float(_req(d, "lat")),
            lon=float(_req(d, "lon")),
            timezone=str(_req(d, "timezone")),
            timezone_offset_sec=tz_off,
            current=CurrentWeather.from_dict(_req(d, "current")),
            minutely=[MinutelyForecast.from_dict(x) for x in d.get("minutely", [])],
            hourly=[HourlyForecast.from_dict(x) for x in d.get("hourly", [])],
            daily=[DailyForecast.from_dict(x) for x in d.get("daily", [])],
        )

    # Convert UTC datetimes to the location's "local" time using timezone_offset (fixed offset)
    @property
    def tzinfo(self) -> timezone:
        return timezone(offset=timezone.utc.utcoffset(None) + self._offset_delta())

    def _offset_delta(self):
        # small helper so we don't import timedelta at top unless needed
        from datetime import timedelta
        return timedelta(seconds=self.timezone_offset_sec)

    def to_local(self, dt_utc: datetime) -> datetime:
        if dt_utc.tzinfo is None:
            raise ValueError("dt_utc must be timezone-aware (UTC).")
        return dt_utc.astimezone(timezone(self._offset_delta()))

    @property
    def current_local_time(self) -> datetime:
        return self.to_local(self.current.dt)

    def to_dict(self) -> Dict[str, Any]:
        """
        Serialize back to a dict. Datetimes are emitted as UNIX seconds to match input shape.
        """
        def encode(obj: Any) -> Any:
            if isinstance(obj, datetime):
                return int(obj.timestamp())
            if isinstance(obj, list):
                return [encode(x) for x in obj]
            if hasattr(obj, "__dict__"):
                return {k: encode(v) for k, v in obj.__dict__.items()}
            return obj

        base = {
            "lat": self.lat,
            "lon": self.lon,
            "timezone": self.timezone,
            "timezone_offset": self.timezone_offset_sec,
            "current": encode(self.current),
            "minutely": encode(self.minutely),
            "hourly": encode(self.hourly),
            "daily": encode(self.daily),
        }
        return base
